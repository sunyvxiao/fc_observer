### 6.2 如果不用 Quest、手动执行时的操作

#### `adapter/platform_detect.py` 核心逻辑：

```python
import sys
import os
from dataclasses import dataclass

@dataclass
class PlatformInfo:
    os_type: str           # "windows" | "linux"
    ebpf_capable: bool
    btf_available: bool
    kernel_version: str

def detect_platform() -> PlatformInfo:
    if sys.platform == "win32":
        return PlatformInfo(
            os_type="windows",
            ebpf_capable=False,
            btf_available=False,
            kernel_version="N/A"
        )
    # Linux
    kernel = os.uname().release
    btf = os.path.exists("/sys/kernel/btf/vmlinux")
    # 检查 libbpf
    try:
        import ctypes
        ctypes.CDLL("libbpf.so.0")
        libbpf_ok = True
    except OSError:
        libbpf_ok = False
  
    return PlatformInfo(
        os_type="linux",
        ebpf_capable=btf and libbpf_ok,
        btf_available=btf,
        kernel_version=kernel
    )
```

#### `adapter/pipe_factory.py` 核心逻辑（Linux FIFO 部分）：

```python
import os
import stat

class LinuxFifoAdapter:
    def __init__(self, config):
        self.event_fifo = config.get("pipeline", {}).get(
            "linux_event_fifo", "/tmp/observer/events")
        self.command_fifo = config.get("pipeline", {}).get(
            "linux_command_fifo", "/tmp/observer/commands")
  
    def _ensure_fifo(self, path):
        dirname = os.path.dirname(path)
        os.makedirs(dirname, exist_ok=True)
        if not os.path.exists(path):
            os.mkfifo(path, mode=0o666)
        elif not stat.S_ISFIFO(os.stat(path).st_mode):
            os.remove(path)
            os.mkfifo(path, mode=0o666)
  
    def open_event_pipe(self, mode="r"):
        self._ensure_fifo(self.event_fifo)
        return open(self.event_fifo, mode)
  
    def open_command_pipe(self, mode="w"):
        self._ensure_fifo(self.command_fifo)
        return open(self.command_fifo, mode)
  
    def cleanup(self):
        for f in [self.event_fifo, self.command_fifo]:
            if os.path.exists(f):
                os.remove(f)
```

#### `adapter/time_source.py` 核心逻辑：

```python
import time
from abc import ABC, abstractmethod

class TimeSource(ABC):
    @abstractmethod
    def now_ns(self) -> int: ...

class RealtimeClock(TimeSource):
    def now_ns(self) -> int:
        return time.time_ns()

class VirtualClock(TimeSource):
    def __init__(self, start_ns=0):
        self._current_ns = start_ns
  
    def now_ns(self) -> int:
        return self._current_ns
  
    def advance(self, delta_ns: int):
        self._current_ns += delta_ns
```
